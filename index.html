let saldoAtual = 10000000.00; let saldoVisivel = true; // 1. Alterar Foto de Perfil via Galeria const fileInput = document.getElementById('file-input'); const profilePic = document.getElementById('profile-pic'); fileInput.addEventListener('change', function(event) { const file = event.target.files[0]; if (file) { const reader = new FileReader(); reader.onload = function(e) { profilePic.src = e.target.result; localStorage.setItem('nu_foto', e.target.result); } reader.readAsDataURL(file); } }); // Carregar dados salvos no localStorage window.addEventListener('DOMContentLoaded', () => { const savedFoto = localStorage.getItem('nu_foto'); if (savedFoto) profilePic.src = savedFoto; const savedSaldo = localStorage.getItem('nu_saldo'); if (savedSaldo !== null) { saldoAtual = parseFloat(savedSaldo); atualizarTelaSaldo(); } }); // 2. Editar Saldo clicando em cima document.getElementById('edit-saldo-btn').addEventListener('click', () => { let novoValor = prompt("Digite o novo valor do saldo:", saldoAtual); if (novoValor !== null && novoValor.trim() !== "") { novoValor = parseFloat(novoValor.replace(/\./g, '').replace(',', '.')); if (!isNaN(novoValor)) { saldoAtual = novoValor; localStorage.setItem('nu_saldo', saldoAtual); atualizarTelaSaldo(); } else { alert("Por favor, digite um número válido."); } } }); function atualizarTelaSaldo() { const spanSaldo = document.getElementById('saldo-value'); if (saldoVisivel) { spanSaldo.innerText = saldoAtual.toLocaleString('pt-BR', { minimumFractionDigits: 2 }); } else { spanSaldo.innerText = "••••"; } } // Ocultar / Mostrar Saldo (Ícone de Olho) document.getElementById('toggle-visibility').addEventListener('click', () => { saldoVisivel = !saldoVisivel; const icon = document.getElementById('toggle-visibility'); if (saldoVisivel) { icon.classList.remove('fa-eye-slash'); icon.classList.add('fa-regular', 'fa-eye'); } else { icon.classList.remove('fa-eye'); icon.classList.add('fa-solid', 'fa-eye-slash'); } atualizarTelaSaldo(); }); // 3. Modais Pix function abrirPixModal() { document.getElementById('pix-modal').style.display = 'flex'; } function fecharPixModal() { document.getElementById('pix-modal').style.display = 'none'; } function realizarPixSimulado() { const nomeDest = document.getElementById('pix-destinatario').value; const valorPix = parseFloat(document.getElementById('pix-valor').value); if (!nomeDest || isNaN(valorPix) || valorPix <= 0) { alert("Preencha todos os campos corretamente."); return; } if (valorPix > saldoAtual) { alert("Saldo insuficiente para esta simulação!"); return; } saldoAtual -= valorPix; localStorage.setItem('nu_saldo', saldoAtual); atualizarTelaSaldo(); const lista = document.getElementById('lista-historico'); const novoItem = document.createElement('div'); novoItem.className = 'historico-item'; const idTransacao = 'E' + Math.floor(Math.random() * 100000000000000); novoItem.onclick = () => gerarComprovante('Pix enviado', `R$ ${valorPix.toFixed(2).replace('.', ',')}`, nomeDest, idTransacao); novoItem.innerHTML = `
Pix enviado para ${nomeDest} Agora mesmo
- R$ ${valorPix.toFixed(2).replace('.', ',')} `; lista.prepend(novoItem); fecharPixModal(); gerarComprovante('Pix enviado', `R$ ${valorPix.toFixed(2).replace('.', ',')}`, nomeDest, idTransacao); } // 4. Comprovante function gerarComprovante(tipo, valor, destinatario, idTransacao) { document.getElementById('comp-valor').innerText = valor; document.getElementById('comp-dest').innerText = destinatario; document.getElementById('comp-id').innerText = idTransacao; document.getElementById('comprovante-modal').style.display = 'flex'; } function fecharComprovante() { document.getElementById('comprovante-modal').style.display = 'none'; }
