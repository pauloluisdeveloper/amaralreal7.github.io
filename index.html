function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var dados = JSON.parse(e.postData.contents); var acao = dados.acao; var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // 1. Ação para gerar o Pix if (acao === "gerar_pix") { var valor = dados.valor || "5,99"; var deviceId = dados.deviceId; // Código Pix Copia e Cola simulado/exemplo var copiaEColaSimulado = "00020126580014br.gov.bcb.pix0136suachavepixaqui5204000053039865802BR5925Djene Teixeira6009Sao Paulo62070503***63041D3D"; return ContentService.createTextOutput(JSON.stringify({ status: "sucesso", copiaECola: copiaEColaSimulado })).setMimeType(ContentService.MimeType.JSON); } // 2. Verificar status do dispositivo if (acao === "verificar") { var deviceId = dados.deviceId; var rows = sheet.getDataRange().getValues(); var liberado = false; for (var i = 1; i < rows.length; i++) { if (rows[i][1] === deviceId && rows[i][3] === "AProvado") { liberado = true; break; } } return ContentService.createTextOutput(JSON.stringify({ status: liberado ? "liberado" : "pendente" })).setMimeType(ContentService.MimeType.JSON); } // 3. Validar Key Manual if (acao === "validar_key") { var keyDigitada = dados.key; var deviceId = dados.deviceId; if (keyDigitada && keyDigitada.trim() !== "") { sheet.appendRow([new Date(), deviceId, keyDigitada, "AProvado", dados.valor || "5,99"]); return ContentService.createTextOutput(JSON.stringify({ status: "sucesso" })).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ status: "erro", mensagem: "Key inválida." })).setMimeType(ContentService.MimeType.JSON); } } // 4. Dados para o Painel Admin if (acao === "admin_dados") { var rows = sheet.getDataRange().getValues(); var totalVendas = 0; var faturamento = 0; var registros = []; for (var i = 1; i < rows.length; i++) { if (rows[i][3] === "AProvado") { totalVendas++; faturamento += parseFloat(rows[i][4] ? rows[i][4].toString().replace(',', '.') : 0); registros.push({ data: rows[i][0] ? Utilities.formatDate(new Date(rows[i][0]), "GMT-3", "dd/MM HH:mm") : "", deviceId: rows[i][1], key: rows[i][2], status: rows[i][3] }); } } return ContentService.createTextOutput(JSON.stringify({ status: "sucesso", totalVendas: totalVendas, faturamento: faturamento.toFixed(2).replace('.', ','), registros: registros.reverse() })).setMimeType(ContentService.MimeType.JSON); } // 5. Registrar clique if (acao === "registrar_clique") { return ContentService.createTextOutput(JSON.stringify({ status: "ok" })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: "erro", mensagem: "Ação desconhecida." })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "erro", mensagem: error.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // Função auxiliar para teste rápido direto no editor, se necessário function doGet(e) { return ContentService.createTextOutput("API do Aplicativo funcionando perfeitamente!").setMimeType(ContentService.MimeType.TEXT); }
